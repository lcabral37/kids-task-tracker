DATA MODEL
──────────
Task {
  id: string                    // Unique identifier
  title: string                 // Task/activity name
  description?: string          // Optional details
  amount: number                // Reward/cost value
  completionCount: number       // Count of completions (0+)
  childId: string               // Which child this belongs to
  date: string                  // "template" or "YYYY-MM-DD"
  type: 'earning' | 'spending'  // Task type
  icon?: string                 // Emoji icon
  templateId?: string           // Points to template if instance
}
